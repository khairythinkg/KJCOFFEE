Panduan Setup Google Apps Script (2 Sheets)

Automatik cipta Sheet 1 (Senarai Tempahan) & Sheet 2 (Ringkasan Bayaran & Resit Drive)

  1. Buka Google Sheets baharu di Google Drive anda.
  2. Di menu atas, tekan Extensions (Peluasan) ➔ Apps Script.
  3. Padam semua kod sedia ada, dan Copy & Paste kod script Dwi-Sheet di bawah:
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    
    // --- SHEET 1: Senarai Tempahan ---
    var sheet1 = ss.getSheetByName("Senarai Tempahan") || ss.insertSheet("Senarai Tempahan");
    if (sheet1.getLastRow() === 0) {
      sheet1.appendRow([
        "Kod Rujukan", "Tarikh Ramadan", "Nama Pelanggan", "No. Phone", 
        "No. Meja", "Jumlah Pax", "Dewasa", "Kanak-Kanak", "Warga Emas", 
        "Masa", "Jumlah (RM)", "Deposit (RM)", "Status", "Catatan", "Masa Dicipta"
      ]);
      sheet1.getRange(1, 1, 1, 15).setFontWeight("bold").setBackground("#0d3b2e").setFontColor("#ffffff");
    }
    
    // --- SHEET 2: Ringkasan Bayaran & Resit ---
    var sheet2 = ss.getSheetByName("Ringkasan Bayaran & Resit") || ss.insertSheet("Ringkasan Bayaran & Resit");
    if (sheet2.getLastRow() === 0) {
      sheet2.appendRow([
        "Nama Pelanggan", "Kod Rujukan", "No. Phone", "Jumlah Bayaran (RM)", 
        "Deposit Dibayar (RM)", "Baki Belum Dibayar (RM)", "Status Bayaran", 
        "Bukti Pembayaran Deposit", "Bukti Pembayaran Penuh", "Tarikh Kemaskini"
      ]);
      sheet2.getRange(1, 1, 1, 10).setFontWeight("bold").setBackground("#d4af37").setFontColor("#ffffff");
    }
    
    // Ekstrak Data
    var refCode = String(data.refCode || "").trim();
    var name = data.name || data.customerName || "";
    var phone = data.phone || "";
    var tableNo = data.tableNo || data.tableName || "";
    var date = data.date || data.dateStr || "";
    var pax = data.pax || data.totalPax || 0;
    var adults = data.adults || data.dewasa || 0;
    var children = data.children || data.kanak || 0;
    var seniors = data.seniors || data.warga || 0;
    var time = data.time || data.arrivalTime || "";
    var totalPrice = Number(data.totalPrice || data.totalAmount || 0);
    var deposit = Number(data.deposit || data.depositAmount || 0);
    var status = String(data.status || "PRE-BOOKED").toUpperCase();
    var notes = data.notes || "";
    var createdAt = data.createdAt || new Date().toLocaleString();
    
    // 1. SIMPAN GAMBAR RESIT KE GOOGLE DRIVE & DAPATKAN LINK
    var receiptLink = "Tiada Resit";
    var base64Data = data.receiptBase64 || data.receiptDataUrl || data.receipt;
    
    if (base64Data && base64Data.indexOf("base64,") !== -1) {
      try {
        var folderName = "Resit Buffet Ramadan 2027";
        var folders = DriveApp.getFoldersByName(folderName);
        var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
        
        var contentType = base64Data.substring(5, base64Data.indexOf(';'));
        var bytes = Utilities.base64Decode(base64Data.substr(base64Data.indexOf('base64,') + 7));
        var blob = Utilities.newBlob(bytes, contentType, "Resit_" + refCode + ".jpg");
        var file = folder.createFile(blob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        receiptLink = file.getUrl();
      } catch(err) {
        receiptLink = "Resit Ada (Gagal Simpan Ke Drive: " + err.toString() + ")";
      }
    } else if (data.hasReceipt) {
      receiptLink = "Resit Dimuat Naik Dalam Web";
    }

    // 2. LOGIK KIRAAN BAKI & STATUS BAYARAN
    var isFullPaid = (status.indexOf("LUNAS") !== -1 || status.indexOf("PAID FULL") !== -1 || status.indexOf("FULL") !== -1);
    var depositPaid = isFullPaid ? totalPrice : deposit;
    var baki = isFullPaid ? 0 : Math.max(0, totalPrice - depositPaid);
    
    var depositProof = receiptLink;
    var fullProof = isFullPaid ? (receiptLink !== "Tiada Resit" ? receiptLink : "Lunas (Bayaran Kaunter/Online)") : "Belum Lunas";
    
    // 3. KEMASKINI / TAMBAH BARIS DI SHEET 1
    var sheet1Data = sheet1.getDataRange().getValues();
    var rowIndex1 = -1;
    for (var i = 1; i < sheet1Data.length; i++) {
      if (String(sheet1Data[i][0]).trim() === refCode) {
        rowIndex1 = i + 1;
        break;
      }
    }
    
    var rowData1 = [
      refCode, date, name, phone, tableNo, pax, adults, children, seniors, time, totalPrice, depositPaid, status, notes, createdAt
    ];
    
    if (rowIndex1 > 0) {
      sheet1.getRange(rowIndex1, 1, 1, rowData1.length).setValues([rowData1]);
    } else {
      sheet1.appendRow(rowData1);
    }
    
    // 4. KEMASKINI / TAMBAH BARIS DI SHEET 2
    var sheet2Data = sheet2.getDataRange().getValues();
    var rowIndex2 = -1;
    for (var j = 1; j < sheet2Data.length; j++) {
      if (String(sheet2Data[j][1]).trim() === refCode) {
        rowIndex2 = j + 1;
        break;
      }
    }
    
    var rowData2 = [
      name, refCode, phone, totalPrice, depositPaid, baki, status, depositProof, fullProof, new Date().toLocaleString()
    ];
    
    if (rowIndex2 > 0) {
      sheet2.getRange(rowIndex2, 1, 1, rowData2.length).setValues([rowData2]);
    } else {
      sheet2.appendRow(rowData2);
    }
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
