Panduan Setup Google Apps Script

Langkah untuk menghubungkan Google Sheet kedai anda

  1. Buka Google Sheets baharu di Google Drive anda (Namakan contoh: Tempahan Ramadan 2027 KJ Lab).
  2. Di menu atas, tekan Extensions (Peluasan) ➔ Apps Script.
  3. Padam semua kod sedia ada, dan Copy & Paste kod script di bawah:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Auto-create Header if sheet empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Ref Code", "Tarikh Ramadan", "Nama Pelanggan", "No Phone", "No Meja", "Jumlah Pax", "Dewasa", "Kanak", "Warga", "Masa", "Jumlah RM", "Deposit RM", "Status", "Catatan", "Kemaskini Terakhir"]);
      sheet.getRange(1, 1, 1, 15).setFontWeight("bold").setBackground("#103e31").setFontColor("#d4af37");
    }
    
    // Check if refCode exists to update status
    var rows = sheet.getDataRange().getValues();
    var rowIndex = -1;
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] === data.refCode) {
        rowIndex = i + 1;
        break;
      }
    }
    
    if (rowIndex > 0) {
      // Update existing row status
      sheet.getRange(rowIndex, 13).setValue(data.status);
      sheet.getRange(rowIndex, 15).setValue(new Date().toLocaleString('ms-MY'));
    } else {
      // Append new reservation row
      sheet.appendRow([
        data.refCode,
        data.dateStr,
        data.customerName,
        "'" + data.phone,
        data.tableName || data.tableCode,
        data.totalPax,
        data.pax ? data.pax.dewasa : "-",
        data.pax ? data.pax.kanak : "-",
        data.pax ? data.pax.warga : "-",
        data.arrivalTime || "6:30 PM",
        data.totalAmount,
        data.depositAmount,
        data.status,
        data.notes || "-",
        new Date().toLocaleString('ms-MY')
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
                         .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()}))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}
